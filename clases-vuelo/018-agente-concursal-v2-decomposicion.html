"""
agente_v2.py — Agente Concursal v2
Arquitectura descompuesta con skills especializados.

La diferencia clave con v1:
- v1: consultar_agente() hace todo en un solo paso
- v2: el orquestador decide qué skill usar según la tarea,
      puede encadenar múltiples skills, y tiene memoria entre sesiones.
"""

import os, json
import anthropic

# Importar los skills especializados
from skills.busqueda   import buscar_contexto_vectorial, buscar_fallback
from skills.calificacion import clasificar_creditos, verificar_prelacion
from skills.redaccion  import redactar_escrito
from memoria.episodica import cargar_sesion, guardar_sesion
from hooks             import before_tool, after_tool, on_error

CLAUDE_MODEL = "claude-opus-4-5"
MAX_TOKENS   = 4096
MAX_ITERACIONES = 5  # límite de seguridad para el loop del agente

# ── Herramientas disponibles para el agente ──────────────────────────
# Claude verá estas definiciones y decidirá cuándo llamar cada una
TOOLS = [
    {
        "name": "buscar_contexto_normativo",
        "description": """Busca en la base de conocimiento (ChromaDB) fragmentos
        relevantes de normas, jurisprudencia y modelos de escritos.
        Usar para fundamentar cualquier escrito o análisis.""",
        "input_schema": {
            "type": "object",
            "properties": {

                "n_fragmentos": {"type": "integer", "default": 5}
            },
            "required": ["consulta"]
        }
    },
    {
        "name": "verificar_prelacion_creditos",
        "description": """Verifica el orden de prelación para una lista de créditos
        según Art. 2495 C.C. y Ley 1116 de 2006. Retorna lista ordenada con clases.""",

# ── Ejecutar una tool cuando Claude la llama ─────────────────────────
def _ejecutar_tool(name: str, params: dict, sesion_id: str) -> str:
    """
    Dispatch de tools. Ejecuta el skill correcto según el nombre.
    Integra hooks: before_tool (antes), after_tool (después), on_error (si falla).
    """
    # Hook: registrar que se está ejecutando una tool
    before_tool(name, params, sesion_id)

    try:
        if name == "buscar_contexto_normativo":
            # Buscar en ChromaDB con fallback graceful
            resultado = buscar_contexto_vectorial(
                params["consulta"],
                n=params.get("n_fragmentos", 5)
            )
            if not resultado:
                # Fallback: responder sin contexto antes que fallar
                resultado = buscar_fallback(params["consulta"])

        elif name == "verificar_prelacion_creditos":
            resultado = verificar_prelacion(params["creditos"])

        elif name == "guardar_preferencia_abogado":
            guardar_sesion(sesion_id, params["clave"], params["valor"])
            resultado = f"Preferencia guardada: {params['clave']} = {params['valor']}"

        else:
            resultado = f"Tool '{name}' no implementada"

        # Hook: registrar el resultado (para auditoría y debugging)
        after_tool(name, params, resultado, sesion_id)
        return resultado

    except Exception as e:
        # Hook: registrar el error, pero NO propagar — el agente debe continuar
        on_error(name, params, e, sesion_id)
        return f"Error en {name}: {str(e)}. Continuar sin este resultado."


# ── El loop del agente v2 ────────────────────────────────────────────
def consultar_agente_v2(mensaje: str, historial: list = None, sesion_id: str = "default"):
    """
    Agente v2 con loop multi-paso.

    A diferencia de v1 que hace UNA llamada a Claude y retorna,
    este agente hace UN LOOP:
    1. Enviar mensaje + tools disponibles a Claude
    2. Si Claude pide una tool: ejecutarla y continuar
    3. Si Claude responde directamente: retornar
    4. Máximo MAX_ITERACIONES para evitar loops infinitos
    """

    if historial is None:
        historial = []

    # Cargar preferencias del abogado (memoria episódica)
    preferencias = cargar_sesion(sesion_id)
    sistema_base = _construir_system_prompt(preferencias)

    cliente = anthropic.Anthropic(api_key=os.environ.get("ANTHROPIC_API_KEY"))

    # Historial de mensajes para ESTA conversación
    mensajes = historial + [{"role": "user", "content": mensaje}]
    fuentes = []

    # ── El loop del agente ──────────────────────────────────────────
    for iteracion in range(MAX_ITERACIONES):

        respuesta = cliente.messages.create(
            model=CLAUDE_MODEL,
            max_tokens=MAX_TOKENS,
            system=sistema_base,
            tools=TOOLS,
            messages=mensajes
        )

        # ── ¿Claude terminó? Retornar la respuesta ──────────────────
        if respuesta.stop_reason == "end_turn":
            texto_final = ""
            for block in respuesta.content:
                if hasattr(block, "text"):
                    texto_final += block.text

            # Actualizar historial y retornar
            historial_nuevo = mensajes + [
                {"role": "assistant", "content": respuesta.content}
            ]
            if len(historial_nuevo) > 20:
                historial_nuevo = historial_nuevo[-20:]

            return texto_final, historial_nuevo, fuentes

        # ── ¿Claude quiere llamar tools? Ejecutarlas ─────────────────
        if respuesta.stop_reason == "tool_use":
            # Agregar respuesta de Claude al historial
            mensajes.append({"role": "assistant", "content": respuesta.content})

            # Ejecutar TODAS las tools que pidió (pueden ser varias)
            resultados_tools = []
            for block in respuesta.content:
                if block.type == "tool_use":
                    resultado = _ejecutar_tool(block.name, block.input, sesion_id)
                    resultados_tools.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": resultado
                    })
                    # Rastrear fuentes usadas
                    if block.name == "buscar_contexto_normativo":
                        fuentes.append(f"ChromaDB:{block.input['consulta']}")

            # Agregar resultados y continuar el loop
            mensajes.append({"role": "user", "content": resultados_tools})
            continue

        # Stop reason inesperado — salir del loop
        break

    # Si llegamos aquí, se agotaron las iteraciones
    return "No se pudo completar la tarea en el límite de iteraciones.", mensajes, fuentes


def _construir_system_prompt(preferencias: dict) -> str:
    """
    Construye el system prompt dinámicamente incluyendo las preferencias
    del abogado guardadas en sesiones anteriores.
    """
    # Importar el SYSTEM_PROMPT base de agente.py v1
    from agente import SYSTEM_PROMPT

    extras = ""
    if preferencias:
        extras = "\n\nPREFERENCIAS DEL ABOGADO (recuerdas de sesiones anteriores):\n"
        for k, v in preferencias.items():
            extras += f"- {k}: {v}\n"

    return SYSTEM_PROMPT + extras
hooks.py — hooks de auditoría
"""
hooks.py — Hooks para el agente v2.
Se ejecutan antes y después de cada tool call.
Útiles para: logging, auditoría, métricas, debugging.
"""

import json, time
from pathlib import Path
from datetime import datetime

LOG_FILE = Path("./logs/agent_tools.jsonl")

def before_tool(name: str, params: dict, sesion_id: str):
    """Se ejecuta ANTES de cada tool call. Loguear para auditoría."""
    LOG_FILE.parent.mkdir(exist_ok=True)
    entry = {
        "ts": datetime.now().isoformat(),
        "evento": "before_tool",
        "tool": name,
        "sesion": sesion_id,
        # NO loguear el contenido de los parámetros — puede tener datos de clientes
        "params_keys": list(params.keys())
    }
    with open(LOG_FILE, "a") as f:
        f.write(json.dumps(entry) + "\n")

def after_tool(name: str, params: dict, resultado: str, sesion_id: str):
    """Se ejecuta DESPUÉS de cada tool call. Registrar duración y éxito."""
    LOG_FILE.parent.mkdir(exist_ok=True)
    entry = {
        "ts": datetime.now().isoformat(),
        "evento": "after_tool",
        "tool": name,
        "sesion": sesion_id,
        "ok": not resultado.startswith("Error"),
        "resultado_len": len(resultado)  # solo el tamaño, no el contenido
    }
    with open(LOG_FILE, "a") as f:
        f.write(json.dumps(entry) + "\n")

def on_error(name: str, params: dict, error: Exception, sesion_id: str):
    """Se ejecuta cuando una tool lanza una excepción."""
    LOG_FILE.parent.mkdir(exist_ok=True)
    entry = {
        "ts": datetime.now().isoformat(),
        "evento": "tool_error",
        "tool": name,
        "sesion": sesion_id,
        "error_type": type(error).__name__,
        "error_msg": str(error)
    }
    with open(LOG_FILE, "a") as f:
        f.write(json.dumps(entry) + "\n")
    # Imprimir en Railway logs para debugging inmediato
    print(f"[HOOK ERROR] Tool {name}: {error}")
memoria/episodica.py — memoria entre sesiones
"""
memoria/episodica.py
Memoria episódica: recuerda preferencias del abogado entre sesiones.
Usa SQLite — sin dependencias extra, ya está en Python.
"""

import sqlite3
from pathlib import Path

DB_PATH = Path("./memoria/episodica.db")

def _get_conn():
    DB_PATH.parent.mkdir(exist_ok=True)
    conn = sqlite3.connect(DB_PATH)
    conn.execute("""
        CREATE TABLE IF NOT EXISTS preferencias (
            sesion_id TEXT,
            clave     TEXT,
            valor     TEXT,
            ts        TEXT DEFAULT (datetime('now')),
            PRIMARY KEY (sesion_id, clave)
        )
    """)
    conn.commit()
    return conn

def cargar_sesion(sesion_id: str) -> dict:
    """Carga todas las preferencias guardadas para esta sesión/abogado."""
    conn = _get_conn()
    rows = conn.execute(
        "SELECT clave, valor FROM preferencias WHERE sesion_id=?",
        (sesion_id,)
    ).fetchall()
    return {r[0]: r[1] for r in rows}

def guardar_sesion(sesion_id: str, clave: str, valor: str):
    """Guarda o actualiza una preferencia."""
    conn = _get_conn()
    conn.execute(
        "INSERT OR REPLACE INTO preferencias (sesion_id, clave, valor) VALUES (?,?,?)",
        (sesion_id, clave, valor)
    )
    conn.commit()

Semana 3: Pasar a tool use estructurado

Separar la búsqueda en ChromaDB como tool formal (con input_schema) en lugar de hacerla siempre. Agregar verificar_prelacion_creditos como segunda tool. Claude ahora puede decidir si buscar o no según el contexto.

4

Semana 4: Loop multi-paso completo

Implementar el loop del agente v2. Para la mayoría de consultas, Claude seguirá respondiendo en un paso. Pero para tareas complejas (proyecto de calificación completo, análisis de acuerdo), el agente podrá buscar → analizar → redactar en múltiples pasos.

5

Semana 5-6: Skills especializados en MCP

Una vez que el loop funciona, separar los skills en el MCP server con el SDK oficial (clase 013). Cada skill se convierte en una MCP tool con TypedDict y tests unitarios.
