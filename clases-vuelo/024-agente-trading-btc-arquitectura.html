2. APIs Gratuitas para BTC

Las mejores opciones sin costo

  • CoinGecko: precio actual, histórico 30 días, market cap, volumen. Sin API key para uso básico. api.coingecko.com/api/v3
  • Binance API: velas (candlesticks) en tiempo real, todos los pares. Sin auth para datos públicos. api.binance.com/api/v3
  • CryptoCompare: noticias + precios. Key gratuita disponible. min-api.cryptocompare.com
  • Alternative.me Fear & Greed Index: índice de sentimiento del mercado 0-100. Gratis. api.alternative.me/fng/

3. Análisis Técnico Básico — Qué es y cómo Claude lo interpreta

No necesitas saber finanzas para entender estos conceptos. Son matemáticas simples aplicadas a precios.

RSI (Relative Strength Index)

Mide si algo está sobrecomprado o sobrevendido. Escala de 0 a 100:

  • RSI < 30: sobrevendido — posiblemente barato, candidato a subir
  • RSI 30-70: zona neutral
  • RSI > 70: sobrecomprado — posiblemente caro, candidato a corregir

Como en un proceso concursal: si todos los acreedores están vendiendo sus créditos al 20% del valor nominal (RSI bajo), puede haber oportunidad.

Media Móvil (MA)

Promedio del precio en los últimos N días. Suaviza el ruido:

  • MA7: promedio 7 días (corto plazo)
  • MA30: promedio 30 días (mediano plazo)
  • Precio > MA30: tendencia alcista reciente
  • MA7 cruza MA30 hacia arriba: "golden cross" — señal de momentum positivo

4. Código Completo — Agente de Análisis BTC

#!/usr/bin/env python3
"""
Agente de Análisis BTC para Quarta
=====================================
Obtiene datos de BTC, calcula indicadores técnicos básicos,
consulta a Claude para análisis, y genera una señal de seguimiento.

Instalación:
  pip install anthropic requests python-dotenv

Variables de entorno (.env):
  ANTHROPIC_API_KEY=sk-ant-...
"""

import os
import json
import requests
from datetime import datetime, timedelta
from dotenv import load_dotenv
import anthropic

# ─── Configuración ─────────────────────────────────────────────
load_dotenv()

# Cliente de Anthropic — usará ANTHROPIC_API_KEY del .env
cliente = anthropic.Anthropic()

# Configuración del agente
MODELO = "claude-opus-4-5"
SIMBOLO = "bitcoin"           # ID de CoinGecko
SIMBOLO_DISPLAY = "BTC/USD"
DIAS_HISTORICO = 30           # días de historial para análisis


# ─── 1. OBTENER DATOS DE MERCADO ────────────────────────────────

def obtener_precio_actual() -> dict:
    """
    Obtiene el precio actual de BTC en USD desde CoinGecko.
    API gratuita, no requiere autenticación.
    """
    url = f"https://api.coingecko.com/api/v3/simple/price"
    params = {
        "ids": SIMBOLO,
        "vs_currencies": "usd",
        "include_24hr_change": "true",
        "include_24hr_vol": "true",
        "include_market_cap": "true"
    }

    try:
        respuesta = requests.get(url, params=params, timeout=10)
        respuesta.raise_for_status()
        datos = respuesta.json()[SIMBOLO]
        return {
            "precio": datos["usd"],
            "cambio_24h": datos["usd_24h_change"],
            "volumen_24h": datos["usd_24h_vol"],
            "market_cap": datos["usd_market_cap"],
            "timestamp": datetime.now().isoformat()
        }
    except requests.RequestException as e:
        print(f"[AGENTE][ERROR] No se pudo obtener precio: {e}")
        return None


def obtener_historico() -> list:
    """
    Obtiene los precios de cierre de los últimos DIAS_HISTORICO días.
    Retorna una lista de floats ordenada del más antiguo al más reciente.
    """
    url = f"https://api.coingecko.com/api/v3/coins/{SIMBOLO}/market_chart"
    params = {
        "vs_currency": "usd",
        "days": DIAS_HISTORICO,
        "interval": "daily"
    }

    try:
        respuesta = requests.get(url, params=params, timeout=10)
        respuesta.raise_for_status()
        datos = respuesta.json()
        # CoinGecko retorna [[timestamp_ms, precio], ...]
        precios = [punto[1] for punto in datos["prices"]]
        return precios
    except requests.RequestException as e:
        print(f"[AGENTE][ERROR] No se pudo obtener histórico: {e}")
        return []


def obtener_fear_greed() -> dict:
    """
    Obtiene el índice Fear & Greed de Alternative.me.
    0-24: Miedo extremo | 25-49: Miedo | 50-74: Codicia | 75-100: Codicia extrema
    """
    try:
        respuesta = requests.get("https://api.alternative.me/fng/", timeout=10)
        respuesta.raise_for_status()
        datos = respuesta.json()["data"][0]
        return {
            "valor": int(datos["value"]),
            "clasificacion": datos["value_classification"],
            "timestamp": datos["timestamp"]
        }
    except Exception as e:
        print(f"[AGENTE][WARN] No se pudo obtener Fear & Greed: {e}")
        return {"valor": 50, "clasificacion": "Neutral", "timestamp": "N/A"}


# ─── 2. CALCULAR INDICADORES TÉCNICOS ───────────────────────────

def calcular_rsi(precios: list, periodo: int = 14) -> float:
    """
    Calcula el RSI (Relative Strength Index).
    Fórmula estándar de Wilder. Retorna valor 0-100.
    """
    if len(precios) < periodo + 1:
        return 50.0  # valor neutral si no hay suficientes datos

    # Calcular cambios diarios
    cambios = [precios[i] - precios[i-1] for i in range(1, len(precios))]

    # Separar ganancias y pérdidas
    ganancias = [c if c > 0 else 0 for c in cambios[-periodo:]]
    perdidas = [-c if c < 0 else 0 for c in cambios[-periodo:]]

    # Promedios
    avg_ganancia = sum(ganancias) / periodo
    avg_perdida = sum(perdidas) / periodo

    if avg_perdida == 0:
        return 100.0

    rs = avg_ganancia / avg_perdida
    rsi = 100 - (100 / (1 + rs))
    return round(rsi, 2)


def calcular_media_movil(precios: list, dias: int) -> float:
    """Calcula la media móvil simple de los últimos N días."""
    if len(precios) < dias:
        return precios[-1] if precios else 0
    return round(sum(precios[-dias:]) / dias, 2)


def preparar_indicadores(precios: list, precio_actual: dict) -> dict:
    """
    Consolida todos los indicadores en un diccionario para enviar a Claude.
    """
    rsi = calcular_rsi(precios)
    ma7 = calcular_media_movil(precios, 7)
    ma30 = calcular_media_movil(precios, 30)
    precio = precio_actual["precio"]

    # Calcular tendencia de precio en los últimos 7 días
    cambio_7d = ((precio - precios[-7]) / precios[-7] * 100) if len(precios) >= 7 else 0

    return {
        "precio_actual_usd": precio,
        "cambio_24h_pct": round(precio_actual["cambio_24h"], 2),
        "cambio_7d_pct": round(cambio_7d, 2),
        "volumen_24h_usd": round(precio_actual["volumen_24h"] / 1e9, 2),  # en billones
        "market_cap_usd_billones": round(precio_actual["market_cap"] / 1e9, 2),
        "rsi_14": rsi,
        "ma_7_dias": ma7,
        "ma_30_dias": ma30,
        "precio_sobre_ma7": precio > ma7,
        "precio_sobre_ma30": precio > ma30,
        "ma7_sobre_ma30": ma7 > ma30,  # golden/death cross
        "max_30d": round(max(precios[-30:]), 2),
        "min_30d": round(min(precios[-30:]), 2),
    }


# ─── 3. ANALIZAR CON CLAUDE ──────────────────────────────────────

def analizar_con_claude(indicadores: dict, fear_greed: dict) -> dict:
    """
    Envía los indicadores a Claude y obtiene un análisis con señal.
    Claude actúa como analista financiero que interpreta datos técnicos.
    """

    prompt = f"""Eres un analista de criptomonedas que interpreta indicadores técnicos.
Analiza los siguientes datos de Bitcoin y genera:
1. Un análisis breve (3-4 oraciones) del estado actual
2. Una señal: HOLD, WATCH o ALERT
3. Los factores más relevantes que justifican la señal

DATOS ACTUALES DE BITCOIN:
{json.dumps(indicadores, indent=2, ensure_ascii=False)}

ÍNDICE FEAR & GREED:
- Valor: {fear_greed['valor']}/100
- Clasificación: {fear_greed['clasificacion']}

DEFINICIÓN DE SEÑALES:
- HOLD: situación estable, sin señales de alarma importante
- WATCH: hay indicadores que merecen atención especial
- ALERT: condición significativa que requiere revisión activa

Responde ÚNICAMENTE en este formato JSON:
{{
  "signal": "HOLD|WATCH|ALERT",
  "analisis": "texto del análisis",
  "factores_clave": ["factor 1", "factor 2", "factor 3"],
  "nivel_confianza": "ALTO|MEDIO|BAJO",
  "nota_riesgo": "recordatorio de que esto NO es consejo financiero"
}}"""

    try:
        respuesta = cliente.messages.create(
            model=MODELO,
            max_tokens=800,
            messages=[{"role": "user", "content": prompt}]
        )

        # Extraer el JSON de la respuesta
        texto = respuesta.content[0].text.strip()
        # Buscar el JSON en el texto
        inicio = texto.find('{')
        fin = texto.rfind('}') + 1
        if inicio >= 0 and fin > inicio:
            return json.loads(texto[inicio:fin])
        else:
            return {"signal": "HOLD", "analisis": texto, "factores_clave": [], "nivel_confianza": "BAJO"}

    except Exception as e:
        print(f"[AGENTE][ERROR] Error en análisis Claude: {e}")
        return {"signal": "HOLD", "analisis": "Error en análisis", "factores_clave": [], "nivel_confianza": "BAJO"}


# ─── 4. FORMATEAR Y MOSTRAR RESULTADO ───────────────────────────

COLORES = {
    "HOLD": "\033[92m",    # verde
    "WATCH": "\033[93m",   # amarillo
    "ALERT": "\033[91m",   # rojo
    "RESET": "\033[0m",
    "GOLD": "\033[33m",
    "DIM": "\033[90m"
}

def mostrar_resultado(indicadores: dict, analisis: dict, fear_greed: dict):
    """
    Imprime el resultado del análisis en la consola con formato legible.
    """
    ts = datetime.now().strftime("%Y-%m-%d %H:%M")
    signal = analisis.get("signal", "HOLD")
    color = COLORES.get(signal, COLORES["RESET"])
    reset = COLORES["RESET"]
    gold = COLORES["GOLD"]
    dim = COLORES["DIM"]

    print(f"\n{gold}{'═'*56}{reset}")
    print(f"{gold}  AGENTE BTC QUARTA · {ts}{reset}")
    print(f"{gold}{'═'*56}{reset}")

    print(f"\n  PRECIO:  ${indicadores['precio_actual_usd']:,.2f} USD")
    print(f"  24h:     {'+' if indicadores['cambio_24h_pct'] > 0 else ''}{indicadores['cambio_24h_pct']:.2f}%")
    print(f"  7d:      {'+' if indicadores['cambio_7d_pct'] > 0 else ''}{indicadores['cambio_7d_pct']:.2f}%")
    print(f"  RSI(14): {indicadores['rsi_14']:.1f}")
    print(f"  MA7:     ${indicadores['ma_7_dias']:,.2f}")
    print(f"  MA30:    ${indicadores['ma_30_dias']:,.2f}")
    print(f"  Fear/Greed: {fear_greed['valor']}/100 ({fear_greed['clasificacion']})")

    print(f"\n  {dim}{'─'*52}{reset}")
    print(f"\n  SEÑAL: {color}[ {signal} ]{reset}  Confianza: {analisis.get('nivel_confianza','N/A')}")
    print(f"\n  {analisis.get('analisis','')}")

    print(f"\n  Factores clave:")
    for f in analisis.get("factores_clave", []):
        print(f"    • {f}")

    print(f"\n  {dim}{analisis.get('nota_riesgo','')}{reset}")
    print(f"\n{gold}{'═'*56}{reset}\n")


# ─── 5. FUNCIÓN PRINCIPAL ───────────────────────────────────────

def ejecutar_agente():
    """
    Punto de entrada principal del agente.
    Orquesta todo el flujo: datos → indicadores → análisis → output.
    """
    print("[AGENTE][INFO] Iniciando análisis BTC...")

    # Obtener todos los datos
    precio_actual = obtener_precio_actual()
    if not precio_actual:
        print("[AGENTE][ERROR] No se pudo obtener el precio. Verifica tu conexión.")
        return

    precios_historicos = obtener_historico()
    if not precios_historicos:
        print("[AGENTE][ERROR] No se pudo obtener el histórico.")
        return

    fear_greed = obtener_fear_greed()

    # Calcular indicadores técnicos
    indicadores = preparar_indicadores(precios_historicos, precio_actual)

    # Análisis con Claude
    print("[AGENTE][INFO] Consultando a Claude para análisis...")
    analisis = analizar_con_claude(indicadores, fear_greed)

    # Mostrar resultado
    mostrar_resultado(indicadores, analisis, fear_greed)

    # Retornar para uso programático
    return {
        "indicadores": indicadores,
        "analisis": analisis,
        "fear_greed": fear_greed
    }


# ─── Ejecutar si es el archivo principal ────────────────────────
if __name__ == "__main__":
    resultado = ejecutar_agente()

5. Paper Trading — Por qué empezar simulado

Un abogado recién graduado no entra directamente al juicio oral. Primero hace audiencias simuladas, trabaja como monitor, observa a otros. El paper trading es lo mismo: aplicas tu estrategia con dinero ficticio. Si funciona durante 3 meses en simulación, entonces consideras mover capital real. 
# paper_trading.py — Extensión del agente para simulación
import json
from datetime import datetime

class PaperTrader:
    """
    Simula operaciones de trading sin dinero real.
    Guarda un historial de señales y "decisiones" para análisis posterior.
    """
    def __init__(self, capital_inicial: float = 1000.0):
        self.capital = capital_inicial
        self.btc = 0.0
        self.historial = []

    def registrar_senal(self, precio: float, senal: str, analisis: str):
        """Registra la señal del agente sin ejecutar nada real."""
        registro = {
            "timestamp": datetime.now().isoformat(),
            "precio_btc": precio,
            "senal": senal,
            "capital_usd": self.capital,
            "btc_en_cartera": self.btc,
            "valor_total": self.capital + (self.btc * precio),
            "analisis_resumen": analisis[:100]
        }
        self.historial.append(registro)

        # Guardar en archivo para análisis futuro
        with open("paper_trading_log.json", "w") as f:
            json.dump(self.historial, f, indent=2, ensure_ascii=False)

        print(f"[PAPER] Señal registrada: {senal} @ ${precio:,.2f}")
        print(f"[PAPER] Valor total simulado: ${registro['valor_total']:,.2f}")
